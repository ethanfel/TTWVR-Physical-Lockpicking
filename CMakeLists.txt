set(OPENXR_INCLUDE_DIR /usr/include/openxr CACHE PATH "Khronos OpenXR headers")
add_library(lockpick SHARED src/Plugin.cpp src/PinFavourite.cpp src/KitRuntime.cpp src/HostInput.cpp src/MotionShaders.cpp src/WorldPin.cpp src/WorldLock.cpp src/NativeScene.cpp src/Inventory.cpp src/Layer.cpp src/Render.cpp src/Video.cpp src/GameCapture.cpp src/Background.cpp src/Target.cpp)
set_target_properties(lockpick PROPERTIES PREFIX "" OUTPUT_NAME TTWVRPhysicalLockpicking)
target_include_directories(lockpick PRIVATE "${OPENXR_INCLUDE_DIR}")
target_compile_definitions(lockpick PRIVATE XR_NO_PROTOTYPES XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11 WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
 target_compile_options(lockpick PRIVATE /W4)
else()
 target_compile_options(lockpick PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers -Wno-cast-function-type)
endif()
target_link_libraries(lockpick PRIVATE d3d11 d3dcompiler dxgi dxguid advapi32 bcrypt)
if(MINGW)
 target_link_options(lockpick PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--kill-at)
endif()
